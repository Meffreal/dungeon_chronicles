# Sim.html — Combat Simulation Tool Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Vytvořit `frontend/sim.html` — standalone Raidbots-style balance tool, čistě frontendový (file://) port combat enginu do JS, s PvP i PvE simulacemi.

**Architecture:** Jeden monolitický HTML soubor se 4 JS sekcemi: sim-data (konstanty), sim-builds (archetypy + enemy configs), sim-engine (port `simulate_unified_combat` do JS), sim-ui (interaktivní UI s detailním logem i hromadnou statistikou). Bez jakéhokoliv backendu — otevřeš v prohlížeči přímo z disku.

**Tech Stack:** Vanilla JS (ES6+), HTML5, CSS3 — žádné závislosti. Formule věrně kopírují `backend/game/combat_engine.py`, `combat_stats.py`, `talents.py`, `models/subclass.py`.

// === SIM-DATA ===
const MAX_ROUNDS = 30;
const CRIT_DAMAGE_MULT = 1.75;
const DAMAGE_VARIANCE = 0.15;
const ABILITY_TRIGGER_PROB = 0.40;
const MAX_CRIT_CHANCE = 0.40;
const T2_COOLDOWN = 4;

const CLASS_WEAPON_BASE = { warrior: 8, ranger: 6, mage: 5 };
const CLASS_ARMOR_CAPS  = { warrior: 0.35, ranger: 0.25, mage: 0.20 };
const CLASS_HP_MULT     = { warrior: 3.5, ranger: 3.5, mage: 3.0 };

function softCapStat(v) {
  if (v <= 50) return v;
  if (v <= 150) return 50 + (v - 50) * 0.70;
  return 50 + 70 + (v - 150) * 0.30;
}
function calcDamageComponents(cls, str_, dex, int_, weaponDmg) {
  let primary, secA, secB;
  if (cls === 'warrior')     { primary = softCapStat(str_); secA = softCapStat(dex)/2; secB = softCapStat(int_)/2; }
  else if (cls === 'ranger') { primary = softCapStat(dex); secA = softCapStat(str_)/2; secB = softCapStat(int_)/2; }
  else if (cls === 'mage')   { primary = softCapStat(int_); secA = softCapStat(str_)/2; secB = softCapStat(dex)/2; }
  else                       { return [weaponDmg, 0, 0]; }
  const base = weaponDmg * (1 + primary / 10) + secA + secB;
  return [base, secA, secB];
}
function calcArmorPct(cls, armorValue, enemyLevel) {
  const cap = CLASS_ARMOR_CAPS[cls] ?? 0.25;
  return Math.min(cap, armorValue / (Math.max(1, enemyLevel) * 10));
}
function calcCritChance(luck, enemyLevel) {
  return Math.min(0.50, luck / (Math.max(1, enemyLevel) * 4));
}
```

function calcDamage(attacker, defender, isCrit = false, dmgOverride = 0, ignoreArmor = false, critMult = CRIT_DAMAGE_MULT) {
  let base = dmgOverride > 0 ? dmgOverride : attacker.effectiveDmg();
  if (!ignoreArmor) {
    const armorPct = calcArmorPct(defender.cfg.cls || '', Math.floor(defender.armor_value * defender.armor_mult), attacker.level);
    base *= (1 - armorPct);
  }
  const variance = 1 + (Math.random() * 2 - 1) * DAMAGE_VARIANCE;
  let dmg = Math.floor(base * variance);
  if (isCrit) dmg = Math.floor(dmg * critMult);
  return Math.max(1, dmg);
}

function applyShieldAbsorb(fighter, rawDmg) {
  const absorb = fighter.shieldAbsorb();
  if (absorb > 0) return Math.max(1, rawDmg - Math.floor(rawDmg * absorb));
  return rawDmg;
}

function processStatusTicks(fighter, other, roundNum, log) {
  const hasPois  = fighter.hasStatus('poison');
  const hasRegen = fighter.hasStatus('regen');
  let poisonSkip = false, regenSkip = false;

  // Interakce: poison + regen neutralizace
  if (hasPois && hasRegen) {
    const ps = fighter.getStatus('poison');
    const rs = fighter.getStatus('regen');
    const net = Math.floor((rs?.tick_value||0) - (ps?.tick_value||0));
    if (net < 0) fighter.hp = Math.max(0, fighter.hp + net);
    else if (net > 0) fighter.hp = Math.min(fighter.hp_max, fighter.hp + net);
    const idef = STATUS_INTERACTIONS['poison+regen'];
    log.push({ type: 'interaction', round: roundNum, actor: fighter.cfg.name, target: fighter.cfg.name,
      damage: net < 0 ? Math.abs(net) : 0, heal: net > 0 ? net : 0,
      actor_hp: fighter.hp, target_hp: fighter.hp, actor_hp_max: fighter.hp_max, target_hp_max: fighter.hp_max,
      status_name: 'poison+regen', status_emoji: idef.emoji,
      text: `  ${idef.emoji} ${fighter.cfg.name}: ${idef.name}! Čistý efekt: ${net >= 0 ? '+' : ''}${net} HP` });
    poisonSkip = true; regenSkip = true;
  }

  const expired = [];
  for (const status of [...fighter.statuses]) {
    const sdef = STATUS_DEFS[status.name] || {};

    if (status.name === 'poison' && poisonSkip) { status.remaining_rounds--; if (status.remaining_rounds <= 0) expired.push(status.name); continue; }
    if (status.name === 'regen'  && regenSkip)  { status.remaining_rounds--; if (status.remaining_rounds <= 0) expired.push(status.name); continue; }

    if (sdef.tick_dmg_pct || sdef.tick_dmg_flat) {
      let dmg = Math.max(1, Math.floor(status.tick_value));
      const emoji = sdef.emoji || '⚡';

      // bleed + weaken → Zranitelnost
      if (status.name === 'bleed' && fighter.hasStatus('weaken')) {
        const amplified = Math.floor(dmg * 1.50);
        const idef = STATUS_INTERACTIONS['bleed+weaken'];
        log.push({ type: 'interaction', round: roundNum, actor: fighter.cfg.name, target: fighter.cfg.name,
          actor_hp: fighter.hp, target_hp: fighter.hp, actor_hp_max: fighter.hp_max, target_hp_max: fighter.hp_max,
          status_emoji: idef.emoji, text: `  ${idef.emoji} ${fighter.cfg.name}: ${idef.name}! Bleed: ${dmg}→${amplified}` });
        dmg = amplified;
      }
      fighter.hp = Math.max(0, fighter.hp - dmg);

      // burn + shield → Tavení Štítu
      if (status.name === 'burn') {
        const sh = fighter.getStatus('shield');
        if (sh) {
          const oldAbs = sh.absorb_pct;
          sh.absorb_pct = Math.max(0.05, sh.absorb_pct - 0.10);
          const idef = STATUS_INTERACTIONS['burn+shield'];
          log.push({ type: 'interaction', round: roundNum, actor: fighter.cfg.name, target: fighter.cfg.name,
            actor_hp: fighter.hp, target_hp: fighter.hp, actor_hp_max: fighter.hp_max, target_hp_max: fighter.hp_max,
            status_emoji: idef.emoji, text: `  ${idef.emoji} ${fighter.cfg.name}: ${idef.name}! Absorpce: ${Math.floor(oldAbs*100)}%→${Math.floor(sh.absorb_pct*100)}%` });
        }
      }

      log.push({ type: 'status_tick', round: roundNum, actor: fighter.cfg.name, target: fighter.cfg.name,
        damage: dmg, actor_hp: fighter.hp, target_hp: fighter.hp, actor_hp_max: fighter.hp_max, target_hp_max: fighter.hp_max,
        status_name: status.name, status_emoji: emoji,
        text: `  ${emoji} ${fighter.cfg.name} trpí ${sdef.name}! -${dmg} HP  [HP: ${fighter.hp}]` });

    } else if (sdef.heal_pct) {
      const heal = Math.max(1, Math.floor(status.tick_value));
      fighter.hp = Math.min(fighter.hp_max, fighter.hp + heal);
      log.push({ type: 'heal', round: roundNum, actor: fighter.cfg.name, target: fighter.cfg.name,
        heal, actor_hp: fighter.hp, target_hp: fighter.hp, actor_hp_max: fighter.hp_max, target_hp_max: fighter.hp_max,
        status_name: status.name, status_emoji: sdef.emoji,
        text: `  ${sdef.emoji} ${fighter.cfg.name} se léčí (${sdef.name}) +${heal} HP  [HP: ${fighter.hp}]` });
    }

    status.remaining_rounds--;
    if (status.remaining_rounds <= 0) expired.push(status.name);
  }

  for (const name of expired) {
    fighter.removeStatus(name);
    const sdef = STATUS_DEFS[name] || {};
    log.push({ type: 'status_expire', round: roundNum, actor: fighter.cfg.name, target: fighter.cfg.name,
      actor_hp: fighter.hp, target_hp: fighter.hp, actor_hp_max: fighter.hp_max, target_hp_max: fighter.hp_max,
      status_name: name, status_emoji: sdef.emoji || '',
      text: `  ${sdef.emoji||''} ${sdef.name||name} na ${fighter.cfg.name} expiroval.` });
  }
}

function checkBossPhases(boss, target, roundNum, log) {
  if (!boss.cfg.is_boss || !(boss.cfg.phases||[]).length) return;
  const hpPct = boss.hp / boss.hp_max;
  for (const phase of (boss.cfg.phases||[])) {
    if (boss.triggered_phases.has(phase.phase_num)) continue;
    if (hpPct <= phase.trigger_hp_pct) {
      boss.triggered_phases.add(phase.phase_num);
      boss.current_phase = phase.phase_num;
      // Stat buffs
      for (const [stat, mult] of Object.entries(phase.stat_multipliers||{})) {
        if (stat === 'atk')       boss.dmg_mult  *= mult;
        else if (stat === 'def')  boss.armor_mult *= mult;
        else if (stat === 'dmg_bonus') boss.dmg_bonus_pct += (mult - 1);
      }
      if (phase.heal_pct > 0) { const h = Math.floor(boss.hp_max * phase.heal_pct); boss.hp = Math.min(boss.hp_max, boss.hp + h); }
      if (phase.apply_status_to_player) target.addStatus(phase.apply_status_to_player, target.hp_max, boss.effectiveAtk());
      log.push({ type: 'phase_change', round: roundNum, actor: boss.cfg.name, target: target.cfg.name,

        phase_num: phase.phase_num, phase_msg: phase.message,
        text: `⚡ PHASE CHANGE! ${phase.message}` });
    }
  }
}

function tryBossSpecial(boss, target, roundNum, log, trigger) {

  for (roundNum = 1; roundNum <= maxRounds; roundNum++) {
    log.push({ type: 'round_start', round: roundNum, actor: '', target: '',
      actor_hp: a.hp, target_hp: b.hp, actor_hp_max: a.hp_max, target_hp_max: b.hp_max,
      text: `\n[Kolo ${roundNum}]` });

    processStatusTicks(a, b, roundNum, log);

        const rAmt = Math.max(1, Math.floor(f.hp_max * 0.005));
        f.hp = Math.min(f.hp_max, f.hp + rAmt);
      }
    }

    if (a.hp <= 0 || b.hp <= 0) break;

    // Boss phase checks
    if (b.cfg.is_boss) checkBossPhases(b, a, roundNum, log);
    if (a.cfg.is_boss) checkBossPhases(a, b, roundNum, log);
    // Boss every_n_rounds specials
    if (a.cfg.is_boss) tryBossSpecial(a, b, roundNum, log, 'every_n_rounds');
    if (b.cfg.is_boss) tryBossSpecial(b, a, roundNum, log, 'every_n_rounds');

      if (second.hp <= 0) break;
    } else {
      executeAttack(first, second, roundNum, log);
      if (second.hp <= 0) { maybeSpiriRevenge(second, first, roundNum, log); break; }
      rangerChainOrMulti(first, second, roundNum, log);
      if (second.hp <= 0) break;
    }
    // Second fighter
    if (second.cfg.cls === 'warrior' && second.rage >= 100) {
      warriorBerserkerBurst(second, first, roundNum, log);
      if (first.hp <= 0) break;
    } else {
      executeAttack(second, first, roundNum, log);
      if (first.hp <= 0) { maybeSpiriRevenge(first, second, roundNum, log); break; }
      rangerChainOrMulti(second, first, roundNum, log);
      if (first.hp <= 0) break;
    }

    // T2 každé 4. kolo
    if (roundNum % T2_COOLDOWN === 0) {
      if (a.talent_t2 && a.hp > 0) { executeT2Ability(a, b, roundNum, log); if (b.hp <= 0) break; }
      if (b.talent_t2 && b.hp > 0) { executeT2Ability(b, a, roundNum, log); if (a.hp <= 0) break; }
    }
  }

  // Výsledek
  let winner, aWon, isDraw = false;
  if (b.hp <= 0) { aWon = true;  winner = 'A'; }
  else if (a.hp <= 0) { aWon = false; winner = 'B'; }
  else {
    const aPct = a.hp / a.hp_max, bPct = b.hp / b.hp_max;
    if (Math.abs(aPct - bPct) < 0.02) { isDraw = true; aWon = false; winner = 'draw'; }
    else { aWon = aPct > bPct; winner = aWon ? 'A' : 'B'; }
  }

  log.push({ type: 'combat_end', round: roundNum, actor: winner === 'A' ? cfgA.name : cfgB.name, target: '',
    actor_hp: a.hp, target_hp: b.hp, actor_hp_max: a.hp_max, target_hp_max: b.hp_max,
    text: winner === 'draw' ? '⏰ Remíza!'
        : `🏆 ${winner === 'A' ? cfgA.name : cfgB.name} zvítězil po ${roundNum} kolech!` });

  return { winner, aWon, isDraw, rounds: roundNum, aHp: a.hp, bHp: b.hp, aHpMax: a.hp_max, bHpMax: b.hp_max, log };
}
```

## Poznámky k implementaci

1. **Pozor na `buildFighterForm`** — funkce používá `fighter-form-a` a `fighter-form-b` jako IDs, takže v `init()` musíš zavolat ji pro oba sufixy. Alternativně refaktoruj aby brala suffix jako parametr.

2. **Boss phases** — `cfgB.phases` je array objektů `{ phase_num, trigger_hp_pct, message, stat_multipliers, apply_status_to_player, heal_pct }`. Enemy presets toto nastavují při `buildEnemy`.

3. **Enemy `cls: ''`** — enemyové nemají třídu, `calcDamageComponents` vrátí `[weapon_dmg, 0, 0]` (fallback větev). `calcArmorPct` použije cap 0.25 pro `cls: ''`.

4. **Mass simulation blokuje UI** — `setTimeout(..., 10)` zajistí překreslení tlačítka před spuštěním smyčky. Pro 10000 simulací může trvat 1-2 sekundy, ale je přijatelné.

5. **Subclass abilities `extra_statuses`** — Python engine má pole `extra_statuses` pro Nekromancer/Elementalista. JS implementace to zahrnuje v Task 8 (`for (const extS of (ability.extra_statuses || []))`). Ujisti se, že `SUBCLASS_ABILITIES` objekty mají `extra_statuses: []` kde je relevantní.
